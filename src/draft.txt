#include <glad/glad.h>
#include <glfw/glfw3.h>
#include <stdio.h>
#include <stdbool.h>

int main (void) {
    
    glfwInit();

    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 3);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 3);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);

    GLFWwindow* window = glfwCreateWindow(800, 800, "Window", NULL, NULL);
    if (window == NULL) {
        printf("Error creating a window!\n");
        glfwTerminate();
        return -1;
    }

    glfwMakeContextCurrent(window);

    gladLoadGL();

    glViewport(0, 0, 800, 800);
    glfwSwapBuffers(window);

    float i = 0.0f;
    bool reverse = false;

    while(!glfwWindowShouldClose(window)) {
        // update input
        glfwPollEvents();
        

        // clean buffer and set window information
        
        if (reverse == true && i <= 0.0f) reverse = false;
        if (reverse == true) i -= 0.000025f;

        if (i  < 1.0f && reverse == false) i += 0.000025f;
        if (i >= 1.0f && reverse == false) reverse = true;

        glClearColor(0.0f, 0.0f, 0.0f + i, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT | GL_STENCIL_BUFFER_BIT);

        // draw() abaixo
        


        // refresh
        glfwSwapBuffers(window);
        glFlush();
    }

    glfwDestroyWindow(window);
    glfwTerminate();
    return 0;
}


#include <stdio.h>
#include <glad/glad.h>
#include <glfw/glfw3.h>

int main (void) {

    glfwInit();

    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 3);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 3);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE); // Core means modern

    // glfwCreateWindow(x size, y size, name, fullscreen?, unknown);
    GLFWwindow* window = glfwCreateWindow(800, 800, "Colorful Window", NULL, NULL);

    if (window == NULL) {
        printf("Failed to create GLFW window\n");
        glfwTerminate();
        return -1;
    }

    glfwMakeContextCurrent(window);

    // invoke glad
    gladLoadGL();
    
    // glViewport(start left, start bottom, finish right, finish top);
    glViewport(0, 0, 800, 800);

    // r , g , b , transparent
    glClearColor(0.50f, 0.10f, 0.05f, 1.0f);
    glClear(GL_COLOR_BUFFER_BIT);

    // refreshing windows
    glfwSwapBuffers(window);


    while(!glfwWindowShouldClose(window)) {
        glfwPollEvents();
    }

    glfwDestroyWindow(window);
    glfwTerminate();
    return 0;
}
